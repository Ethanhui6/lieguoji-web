import { readFileSync, writeFileSync } from 'node:fs';
import path from 'node:path';
import { createHash } from 'node:crypto';
const directory = process.argv[2];
if (!/^\.autopilot\/ph06\/playtest-browser(?:-[a-z0-9-]+)?$/.test(directory ?? '')) throw new Error('Use one actual fixed-build playtest evidence directory.');
const bytes = readFileSync(path.join(directory, 'results.json')), report = JSON.parse(bytes.toString('utf8'));
const run = JSON.parse(readFileSync(path.join(directory, 'run.json'), 'utf8'));
const observations = [];
function visit(suite) {
  for (const spec of suite.specs ?? []) for (const test of spec.tests ?? []) for (const result of test.results) for (const attachment of result.attachments ?? []) if (attachment.name === 'generation-progress-response.json') {
    if (typeof attachment.body !== 'string') throw new Error('Missing retained raw DOM observation bytes');
    const evidence = JSON.parse(Buffer.from(attachment.body, 'base64').toString('utf8'));
    for (const raw of evidence.runs) {
      const samples = [{ at: raw.startedAt, stage: '[trusted-preview-click]' }, ...raw.samples, ...(raw.finishedAt === null ? [] : [{ at: raw.finishedAt, stage: `[ui-${raw.finishedState}]` }])];
      const gaps = samples.slice(1).map((sample, index) => ({ ms: sample.at - samples[index].at, from: samples[index].stage, to: sample.stage }));
      const valid = raw.trusted === true && raw.seenBusy === true && raw.finishedAt !== null && raw.finishedState === 'ready' && raw.samples.length > 0
        && samples.every(sample => Number.isFinite(sample.at)) && gaps.every(gap => gap.ms >= 0);
      const maxGap = gaps.length ? gaps.reduce((a, b) => a.ms >= b.ms ? a : b) : null;
      const cancelMs = raw.cancelRequestedAt === null || raw.finishedAt === null ? null : raw.finishedAt - raw.cancelRequestedAt;
      observations.push({ title: spec.title, project: test.projectName, operation: raw.operation, valid, size: spec.title.includes('真实 512') ? 512 : 256,
        durationMs: raw.finishedAt === null ? null : raw.finishedAt - raw.startedAt, maxGap,
        progress250msPassed: valid && maxGap !== null && maxGap.ms <= 250, cancellationMs: cancelMs,
        cancellation500msPassed: cancelMs === null ? null : valid && raw.cancelTrusted === true && cancelMs >= 0 && cancelMs <= 500, raw });
    }
  }
  for (const child of suite.suites ?? []) visit(child);
}
for (const suite of report.suites) visit(suite);
const cancellations = observations.filter(entry => entry.cancellationMs !== null);
const groups = ['desktop-chromium', 'mobile-chromium', 'tablet-chromium'].flatMap(project => [256, 512].flatMap(size => {
  const entries = observations.filter(entry => entry.project === project && entry.size === size); if (!entries.length) return [];
  const cancelled = entries.filter(entry => entry.cancellationMs !== null);
  return [{ project, size, runs: entries.length, validRuns: entries.filter(entry => entry.valid).length, maximumProgressGapMs: Math.max(...entries.map(entry => entry.maxGap?.ms ?? Infinity)),
    cancellationRuns: cancelled.length, maximumCancellationMs: cancelled.length ? Math.max(...cancelled.map(entry => entry.cancellationMs)) : null }];
}));
const summary = { createdAt: new Date().toISOString(), expectedBuildSha256: run.expectedBuildSha256,
  resultsSha256: createHash('sha256').update(bytes).digest('hex'), functionalPassed: run.functionalPassed,
  measurement: 'Read-only actual DOM progress and trusted input observations during Playwright tracing. Desktop hardware, with emulated phone/tablet viewports; not real devices or isolated frame performance.',
  timingScope: '65k macro creation on three projects and 512 macro creation on desktop only. No 250/500 timing claim is made for the separate island/coast/constraint scenarios.',
  expectedProgressRuns: 11, actualProgressRuns: observations.length, expectedCancellationRuns: 4, actualCancellationRuns: cancellations.length,
  observationComplete: observations.length === 11 && cancellations.length === 4 && observations.every(entry => entry.valid),
  progress250msPassed: observations.length === 11 && observations.every(entry => entry.progress250msPassed),
  cancellation500msPassed: cancellations.length === 4 && cancellations.every(entry => entry.cancellation500msPassed), groups,
  largestGaps: observations.toSorted((a, b) => (b.maxGap?.ms ?? Infinity) - (a.maxGap?.ms ?? Infinity)).slice(0, 8).map(({ raw: _raw, ...entry }) => entry),
};
writeFileSync(path.join(directory, 'progress-raw-review.json'), JSON.stringify({ observations }, null, 2) + '\n', { flag: 'wx' });
writeFileSync(path.join(directory, 'progress-review.json'), JSON.stringify(summary, null, 2) + '\n', { flag: 'wx' });
console.log(JSON.stringify(summary, null, 2));
