import fs from 'node:fs';
import path from 'node:path';
import { createHash } from 'node:crypto';
import { fileURLToPath } from 'node:url';

const target = path.dirname(fileURLToPath(import.meta.url));
const root = path.resolve(target, '../../../..');
const source = '.autopilot/ph06/coast-curves-staging';
const hash = bytes => createHash('sha256').update(bytes).digest('hex');
const assert = (ok, why) => { if (!ok) throw new Error(why); };
const bytes = relative => fs.readFileSync(path.join(root, relative));
const rawCopied = [];
const safe = relative => assert(!path.isAbsolute(relative) && !relative.includes('..') && !relative.includes('\\'), `Unsafe relative path ${relative}`);
const save = (relative, data) => {
  safe(relative); const destination = path.join(target, relative);
  fs.mkdirSync(path.dirname(destination), { recursive: true }); fs.writeFileSync(destination, data, { flag: 'wx' });
};
const json = (relative, value) => save(relative, JSON.stringify(value, null, 2) + '\n');
const copy = relative => {
  safe(relative); const original = `${source}/${relative}`, data = bytes(original);
  save(`original/${relative}`, data);
  rawCopied.push({ path: `original/${relative}`, original, sha256: hash(data), bytes: data.length });
};
const walk = relative => {
  const directory = path.join(root, source, relative);
  return fs.readdirSync(directory, { withFileTypes: true }).flatMap(entry => {
    assert(!entry.isSymbolicLink(), 'Source archive must not follow links');
    const file = relative ? `${relative}/${entry.name}` : entry.name;
    return entry.isDirectory() ? walk(file) : [file];
  });
assert(!fs.existsSync(path.join(target, 'verification.json')) && !fs.existsSync(path.join(target, 'original')), 'Never overwrite original evidence');

for (const relative of [...walk('checks/first'), ...walk('full-checks/first'), ...walk('output/first')]) copy(relative);
for (const relative of ['OBSERVED_FIRST.md', 'OBSERVED_FULL_FIRST.md', 'promotion.json', 'static-manifest.json', 'full-views.json']) copy(relative);

const packetVariants = [];
for (const variant of ['baseline', 'candidate']) {
  const prefix = `full-output/first/${variant}`;
  for (const name of ['capture.json', 'outline.json', 'result.json', 'source-before.json', 'source-after.json', 'fine/index.json',
    `${variant}-full-physical.png`, `${variant}-full-coverage.png`, `${variant}-inner-sea-physical.png`, `${variant}-inner-sea-coverage.png`]) copy(`${prefix}/${name}`);
  // Both snapshots are small source/harness closures, not geography packets.
  for (const relative of walk(`${prefix}/source`)) copy(relative);
  const fineRelative = `${source}/${prefix}/fine`;
  const entries = fs.readdirSync(path.join(root, fineRelative), { withFileTypes: true });
  assert(entries.every(entry => entry.isFile() && !entry.isSymbolicLink() && (entry.name === 'index.json' || /^coast-\d+-\d+\.bin$/.test(entry.name))), 'Unexpected packet directory entry');
  const indexBytes = bytes(`${fineRelative}/index.json`), declared = JSON.parse(indexBytes);
  const actual = entries.filter(entry => entry.name.endsWith('.bin')).map(entry => {
    const original = `${fineRelative}/${entry.name}`, data = bytes(original);
    return { file: `fine/${entry.name}`, original, bytes: data.length, sha256: hash(data) };
  }).sort((left, right) => left.file.localeCompare(right.file));
  const actualMap = new Map(actual.map(row => [row.file, row]));
  assert(actual.length === declared.packets.length && actualMap.size === actual.length, 'Packet enumeration and original index count differ');
  const keys = new Set(), listedFiles = new Set(), digest = createHash('sha256');
  let ownerNodes = 0;
  for (const packet of declared.packets) {
    assert(!keys.has(packet.key) && !listedFiles.has(packet.file), 'Duplicate original packet index item'); keys.add(packet.key); listedFiles.add(packet.file);
    const row = actualMap.get(packet.file);
    assert(row && row.bytes === packet.bytes && row.sha256 === packet.sha256, `Actual packet differs from original index: ${variant}/${packet.file}`);
    const data = bytes(row.original); assert(hash(data) === row.sha256, 'Packet changed during digest check');
    digest.update(packet.key).update(data); ownerNodes += packet.ownerNodes;
  }
  const packetDigest = digest.digest('hex');
  assert(packetDigest === declared.packetDigest && ownerNodes === declared.fineNodes, 'Original generation packet digest or owner-node count differs');
  packetVariants.push({ variant, originalDirectory: path.join(root, fineRelative), originalDirectoryRelative: fineRelative,
    packetCount: actual.length, packetBytes: actual.reduce((total, row) => total + row.bytes, 0), ownerNodes,
    originalIndexSha256: hash(indexBytes), originalIndexArchivedAt: `original/${prefix}/fine/index.json`,
    originalOrderedPacketDigest: packetDigest, actualEnumerationMatchesOriginalIndex: true, rawPacketsCopied: false, files: actual });
}
json('omitted-fine-packets.inventory.json', { recordedAt: new Date().toISOString(), enumeration: 'actual regular .bin files, rehashed and cross-checked against original fine/index.json',
  packetPayloadsIncluded: false, independentlyReplayableArchive: false,
  limitation: 'Packet payloads remain only in the listed local original directories. This archive retains hashes, sizes, original indices and selected rendered evidence; it cannot independently replay or decode the full generated fine world.', variants: packetVariants });

for (const row of rawCopied) {
  assert(hash(bytes(row.original)) === row.sha256, `Original changed during archive: ${row.original}`);
  const copied = fs.readFileSync(path.join(target, row.path)); assert(copied.length === row.bytes && hash(copied) === row.sha256, `Copied bytes differ: ${row.path}`);
}
const first = JSON.parse(bytes(`${source}/checks/first/verification.json`));
const full = JSON.parse(bytes(`${source}/full-checks/first/verification.json`));
const inventoryRows = [...rawCopied, ...['archive.mjs', 'omitted-fine-packets.inventory.json'].map(file => {
  const data = fs.readFileSync(path.join(target, file)); return { path: file, sha256: hash(data), bytes: data.length };
})].sort((a, b) => a.path.localeCompare(b.path));
json('files.sha256.json', { excludes: ['files.sha256.json', 'verification.json'], files: inventoryRows });
const inventoryBytes = fs.readFileSync(path.join(target, 'files.sha256.json'));
const verification = {
  archivedAt: new Date().toISOString(), sourceDirectory: source, selectiveArchive: true, originalBytePreserving: true,
  copiedOriginalFiles: rawCopied.length, copiedOriginalBytes: rawCopied.reduce((total, row) => total + row.bytes, 0),
  inventoryFiles: inventoryRows.length, inventorySha256: hash(inventoryBytes),
  originalFirstVerificationSha256: hash(bytes(`${source}/checks/first/verification.json`)),
  originalFullVerificationSha256: hash(bytes(`${source}/full-checks/first/verification.json`)),
  originalFirstPassed: first.passed ?? null, originalFullPassed: full.passed ?? null,
  finePackets: packetVariants.map(({ variant, packetCount, packetBytes, ownerNodes, originalOrderedPacketDigest, originalDirectoryRelative }) => ({ variant, packetCount, packetBytes, ownerNodes, originalOrderedPacketDigest, originalDirectoryRelative })),
  totalFinePacketFiles: packetVariants.reduce((total, row) => total + row.packetCount, 0),
  totalFinePacketBytesNotCopied: packetVariants.reduce((total, row) => total + row.packetBytes, 0),
  rawPacketCopies: 0, independentlyReplayableArchive: false,
  selectedImages: rawCopied.filter(row => row.path.endsWith('.png')).map(row => row.path),
  omitted: ['Full generated fine .bin payloads', 'Raw .i16/.i32/.u32 view and geographic arrays', 'Additional east/west range rendered images'],
  existingResultClaimsPreserved: true, testsRerun: false, buildRun: false, browserRun: false, visualAcceptanceAdded: false, phaseAcceptanceAdded: false, cloudWrites: false,
};
json('verification.json', verification);
console.log(JSON.stringify({ copiedOriginalFiles: verification.copiedOriginalFiles, copiedOriginalBytes: verification.copiedOriginalBytes,
  totalFinePacketFiles: verification.totalFinePacketFiles, totalFinePacketBytesNotCopied: verification.totalFinePacketBytesNotCopied,
  selectedImages: verification.selectedImages.length, inventorySha256: verification.inventorySha256,
  verificationSha256: hash(fs.readFileSync(path.join(target, 'verification.json'))), testsRerun: false, independentlyReplayableArchive: false }));
