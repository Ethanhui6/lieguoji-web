interface SavedManifest {
  version: number; revision: number; atlasId: string;
  spec: { columns: number; rows: number; width: number; height: number; coordinateScale: number; blockCells: number };
  blockRefs: { key: string; hash: string }[];
  generation: { lastApplied: { recipe: { generator: { id: string; version: number; fingerprint: string }; seed?: string; parameters: Record<string, number> }; appliedRevision: number } | null };
}
interface SavedSnapshot {
  rootBytes: number[]; manifestBytes: number[];
  root: { revision: number; domain: string; rootDigest: string; currentManifest: string; backupManifest: string | null };
  manifest: SavedManifest;
}
interface Geography { heights: number[]; xyQ: number[]; verifiedBlocks: number }
interface Audit {
  ok: boolean; manifestDigest: string; draftId: string; revision: number; verifiedBlocks: number;
  display: { source: string; manifestDigest: string; revision: number; version: number; verifiedBlocks: number };
}
interface Metrics {

  lod: string; cellPixels: number; gridVisible: boolean; completedFrameSequence: number; displaySource: string;
  acceptedRevision: number; committedRevision: number; displayRevision: number;
}
async function saved(page: Page, draftId: string): Promise<SavedSnapshot> {
  return page.evaluate(id => new Promise<SavedSnapshot>((resolve, reject) => {
    const request = indexedDB.open('lieguoji-atlas-drafts-v2', 1);
    request.onupgradeneeded = () => request.transaction!.abort(); request.onerror = () => reject(request.error);
    request.onsuccess = () => {
      const db = request.result, tx = db.transaction(['roots', 'manifests'], 'readonly'); let result: SavedSnapshot;
      const root = tx.objectStore('roots').get(id);
      root.onsuccess = () => {
        if (!(root.result instanceof Uint8Array)) { tx.abort(); return; }
        const value = JSON.parse(new TextDecoder().decode(root.result)) as SavedSnapshot['root'];
        const read = tx.objectStore('manifests').get(value.currentManifest);
        read.onsuccess = () => {
          if (!(read.result?.bytes instanceof Uint8Array)) { tx.abort(); return; }
          result = { root: value, rootBytes: [...root.result], manifestBytes: [...read.result.bytes], manifest: JSON.parse(new TextDecoder().decode(read.result.bytes)) };
        };
      };
      tx.oncomplete = () => { db.close(); resolve(result); }; tx.onabort = () => { db.close(); reject(tx.error ?? new Error('Missing saved draft')); };
    };
  }), draftId);
}
async function databaseDigest(page: Page) {
  return page.evaluate(async () => {
    const rows = await new Promise<{ store: string; keys?: IDBValidKey[]; values?: unknown[] }[]>((resolve, reject) => {
      const request = indexedDB.open('lieguoji-atlas-drafts-v2', 1);
      request.onupgradeneeded = () => request.transaction!.abort(); request.onerror = () => reject(request.error);
      request.onsuccess = () => {
        const db = request.result, names = [...db.objectStoreNames].sort(), tx = db.transaction(names, 'readonly'), values = names.map(store => ({ store })) as { store: string; keys?: IDBValidKey[]; values?: unknown[] }[];
        for (const row of values) {
          const store = tx.objectStore(row.store), keys = store.getAllKeys(), records = store.getAll();
          keys.onsuccess = () => { row.keys = keys.result; }; records.onsuccess = () => { row.values = records.result; };
        }
        tx.oncomplete = () => { db.close(); resolve(values); }; tx.onabort = () => { db.close(); reject(tx.error); };
      };
    });
    const bytes = new TextEncoder().encode(JSON.stringify(rows, (_key, value) => value instanceof Uint8Array ? [...value] : value));
    return { bytes: bytes.length, sha256: [...new Uint8Array(await crypto.subtle.digest('SHA-256', bytes))].map(value => value.toString(16).padStart(2, '0')).join('') };
  });
}
async function geography(page: Page, manifest: SavedManifest): Promise<Geography> {
  return page.evaluate(async ({ spec, refs }) => {
    const records = await new Promise<{ key: string; hash: string; bytes: Uint8Array }[]>((resolve, reject) => {
      const request = indexedDB.open('lieguoji-atlas-drafts-v2', 1);
      request.onupgradeneeded = () => request.transaction!.abort(); request.onerror = () => reject(request.error);
      request.onsuccess = () => {
        const db = request.result, tx = db.transaction('blocks', 'readonly'), records: { key: string; hash: string; bytes: Uint8Array }[] = [];
        for (const ref of refs) {
          const read = tx.objectStore('blocks').get(ref.hash);
          read.onsuccess = () => { if (!(read.result?.bytes instanceof Uint8Array)) tx.abort(); else records.push({ ...ref, bytes: read.result.bytes }); };
        }
        tx.oncomplete = () => { db.close(); resolve(records); }; tx.onabort = () => { db.close(); reject(tx.error ?? new Error('Missing geographic block')); };
      };
    });
    const layers = ['geometry', 'elevation', 'sim-membership', 'admin-membership'];
    const seenKeys = new Set<string>(), vertexSeen = new Uint8Array((spec.columns + 1) * (spec.rows + 1)), cellSeen = new Uint8Array(spec.columns * spec.rows);
    const heights = new Int32Array(cellSeen.length), xyQ = new Int32Array(vertexSeen.length * 2), step = spec.width * spec.coordinateScale / spec.columns;
    for (const { key, hash, bytes } of records) {
      const actualHash = [...new Uint8Array(await crypto.subtle.digest('SHA-256', new Uint8Array(bytes).buffer))].map(value => value.toString(16).padStart(2, '0')).join('');
      if (hash !== actualHash) throw new Error('Persisted full-block hash mismatch');
      const [layer, rawX, rawY] = key.split(':'), bx = Number(rawX), by = Number(rawY), view = new DataView(bytes.buffer, bytes.byteOffset, bytes.byteLength);
      if (seenKeys.has(key) || bytes.length < 28 || [76, 71, 65, 66].some((value, index) => bytes[index] !== value)
        || view.getUint16(4, true) !== 1 || layers[bytes[6]] !== layer || bytes[7] !== (layer === 'geometry' ? 2 : 1)
        || view.getUint16(8, true) !== bx || view.getUint16(10, true) !== by || view.getUint32(12, true) !== 0) throw new Error('Invalid LGAB identity/header');
      seenKeys.add(key); let offset = 16; const columns: number[][] = [];
      for (let column = 0; column < bytes[7]; column++) {
        const signed = layer === 'elevation' || layer === 'geometry' && column === 1, length = view.getUint32(offset + 4, true), byteLength = view.getUint32(offset + 8, true);
        if (bytes[offset] !== (signed ? 2 : 1) || bytes[offset + 1] || bytes[offset + 2] || bytes[offset + 3] || length > 2178 || byteLength !== length * 4 || offset + 12 + byteLength > bytes.length) throw new Error('Invalid LGAB column');
        offset += 12; const values: number[] = [];
        for (let slot = 0; slot < length; slot++, offset += 4) values.push(signed ? view.getInt32(offset, true) : view.getUint32(offset, true));
        columns.push(values);
      }
      if (offset !== bytes.length) throw new Error('Unexpected LGAB trailing bytes');
      if (layer === 'geometry') {
        const startX = bx * 32 + Number(bx !== 0), startY = by * 32 + Number(by !== 0), endX = (bx + 1) * 32, endY = (by + 1) * 32;
        if (columns[0].length !== (endX - startX + 1) * (endY - startY + 1) || columns[1].length !== columns[0].length * 2) throw new Error('Wrong owned-vertex coverage');
        let slot = 0;
        for (let row = startY; row <= endY; row++) for (let col = startX; col <= endX; col++, slot++) {
          const id = row * (spec.columns + 1) + col, x = columns[1][slot * 2], y = columns[1][slot * 2 + 1];
          if (columns[0][slot] !== id || vertexSeen[id] || Math.abs(x - col * step) > step / 4 || Math.abs(y - row * step) > step / 4
            || (col === 0 || col === spec.columns) && x !== col * step || (row === 0 || row === spec.rows) && y !== row * step) throw new Error('Changed identity, duplicate owner or invalid vertex bound');
          vertexSeen[id] = 1; xyQ[id * 2] = x; xyQ[id * 2 + 1] = y;
        }
      } else {
        if (columns[0].length !== 1024) throw new Error('Incomplete cell layer');
        if (layer === 'elevation') for (let slot = 0; slot < 1024; slot++) {
          const id = (by * 32 + Math.floor(slot / 32)) * spec.columns + bx * 32 + slot % 32, value = columns[0][slot];
          if (cellSeen[id] || value < -12000 || value > 12000) throw new Error('Invalid real elevation coverage');
          cellSeen[id] = 1; heights[id] = value;
        }
      }
    }
    for (const layer of layers) for (let by = 0; by < spec.rows / 32; by++) for (let bx = 0; bx < spec.columns / 32; bx++) if (!seenKeys.has(`${layer}:${bx}:${by}`)) throw new Error('Incomplete full geography');
    if (vertexSeen.some(value => value !== 1) || cellSeen.some(value => value !== 1) || records.length !== 4 * (spec.columns / 32) * (spec.rows / 32)) throw new Error('Missing real cells/vertices');
    return { heights: [...heights], xyQ: [...xyQ], verifiedBlocks: records.length };
  }, { spec: manifest.spec, refs: manifest.blockRefs });
}
function inspect(data: Geography, spec: SavedManifest['spec']) {
  const { heights, xyQ } = data, adjacent = (id: number) => [id >= spec.columns ? id - spec.columns : -1, id % spec.columns ? id - 1 : -1,
    id % spec.columns + 1 < spec.columns ? id + 1 : -1, id + spec.columns < heights.length ? id + spec.columns : -1].filter(id => id >= 0);
  const coast = new Map<number, number[]>(), edgePairs: [number, number][] = [], areas: number[] = [], seen = new Uint8Array(heights.length), componentCells: number[][] = [];
  let worldAreaQ2 = 0, landAreaQ2 = 0, minCross = Infinity;
  function edge(a: number, b: number) { edgePairs.push([a, b]); coast.set(a, [...coast.get(a) ?? [], b]); coast.set(b, [...coast.get(b) ?? [], a]); }
  for (let id = 0; id < heights.length; id++) {
    const col = id % spec.columns, row = Math.floor(id / spec.columns), a = row * (spec.columns + 1) + col, ids = [a, a + 1, a + spec.columns + 2, a + spec.columns + 1]; let area = 0;
    for (let index = 0; index < 4; index++) {
      const p = ids[index] * 2, q = ids[(index + 1) % 4] * 2, r = ids[(index + 2) % 4] * 2;
      const cross = (xyQ[q] - xyQ[p]) * (xyQ[r + 1] - xyQ[q + 1]) - (xyQ[q + 1] - xyQ[p + 1]) * (xyQ[r] - xyQ[q]);
      if (!(cross > 0)) throw new Error(`Actual quad ${id} lost strict convexity`); minCross = Math.min(minCross, cross);
      area += xyQ[p] * xyQ[q + 1] - xyQ[q] * xyQ[p + 1];
    }
    if (!Number.isSafeInteger(area) || area <= 0) throw new Error('Invalid independently measured quad area');
    areas[id] = area; worldAreaQ2 += area; if (heights[id] >= 0) landAreaQ2 += area;
    if (col + 1 < spec.columns && (heights[id] >= 0) !== (heights[id + 1] >= 0)) edge(a + 1, a + spec.columns + 2);
    if (row + 1 < spec.rows && (heights[id] >= 0) !== (heights[id + spec.columns] >= 0)) edge(a + spec.columns + 1, a + spec.columns + 2);
    if (heights[id] >= 0 && !seen[id]) {
      const queue = [id]; seen[id] = 1;
      for (let head = 0; head < queue.length; head++) for (const next of adjacent(queue[head])) if (!seen[next] && heights[next] >= 0) { seen[next] = 1; queue.push(next); }
      componentCells.push(queue);
    }
  }
  let coastLengthWorld = 0, coastTurnEnergyQ2 = 0, focusEnergy = -1, focusCell = -1;
  for (const [a, b] of edgePairs) coastLengthWorld += Math.hypot(xyQ[a * 2] - xyQ[b * 2], xyQ[a * 2 + 1] - xyQ[b * 2 + 1]) / spec.coordinateScale;
  for (const [id, neighbors] of coast) if (neighbors.length === 2) {
    const [a, b] = neighbors, energy = (xyQ[a * 2] + xyQ[b * 2] - 2 * xyQ[id * 2]) ** 2 + (xyQ[a * 2 + 1] + xyQ[b * 2 + 1] - 2 * xyQ[id * 2 + 1]) ** 2;
    coastTurnEnergyQ2 += energy;
    const col = id % (spec.columns + 1), row = Math.floor(id / (spec.columns + 1));
    if (energy > focusEnergy && col > 0 && col < spec.columns && row > 0 && row < spec.rows) {
      const incident = [(row - 1) * spec.columns + col - 1, (row - 1) * spec.columns + col, row * spec.columns + col - 1, row * spec.columns + col];
      const land = incident.find(cell => heights[cell] >= 0); if (land !== undefined) { focusEnergy = energy; focusCell = land; }
    }
  }
  return { componentCells, componentAreasQ2: componentCells.map(cells => cells.reduce((sum, id) => sum + areas[id], 0)), worldAreaQ2, landAreaQ2, minCross,
    coast, coastlineEdges: edgePairs.length, coastLengthWorld, coastTurnEnergyQ2, focusCell };
}
async function audit(page: Page): Promise<Audit> {
  return page.getByTestId('atlas-canvas').evaluate(element => new Promise<Audit>((resolve, reject) => {
    const requestId = crypto.randomUUID(), timer = setTimeout(() => { element.removeEventListener('atlas-audit-response', listener); reject(new Error('Missing read-only audit')); }, 30000);
    const listener = (event: Event) => { const detail = (event as CustomEvent).detail; if (detail.requestId !== requestId) return;
      clearTimeout(timer); element.removeEventListener('atlas-audit-response', listener); if (!detail.ok) reject(new Error(detail.error)); else resolve(detail); };
    element.addEventListener('atlas-audit-response', listener); element.dispatchEvent(new CustomEvent('atlas-audit-request', { detail: { requestId } }));
  }));
}
async function metrics(page: Page): Promise<Metrics> {

    const requestId = crypto.randomUUID(), timer = setTimeout(() => { element.removeEventListener('atlas-metrics-response', listener); reject(new Error('Missing read-only metrics')); }, 10000);
    const listener = (event: Event) => { const detail = (event as CustomEvent).detail; if (detail.requestId !== requestId) return;
      clearTimeout(timer); element.removeEventListener('atlas-metrics-response', listener); resolve(detail.metrics); };
    element.addEventListener('atlas-metrics-response', listener); element.dispatchEvent(new CustomEvent('atlas-metrics-request', { detail: { requestId } }));
  }));
}

async function refreshInNewDocument(page: Page, draftId: string, expected: SavedSnapshot, errors: string[]) {
  // Keep the original session's intentional in-memory undo history, while a
  // second real document and refresh must reconstruct the durable result.
  const reopened = await page.context().newPage(); reopened.on('pageerror', error => errors.push(error.message));
  try {
    await reopened.goto(page.url()); await ready(reopened); await reopened.reload(); await ready(reopened);
    expect(await saved(reopened, draftId)).toEqual(expected);
    expect((await audit(reopened)).display).toMatchObject({ source: 'committed', revision: expected.root.revision, version: 2, manifestDigest: expected.root.currentManifest, verifiedBlocks: 256 });
    return await geography(reopened, expected.manifest);
  } finally { await reopened.close(); }
}
function collectErrors(page: Page) {
  const errors: string[] = []; page.on('pageerror', error => errors.push(error.message)); page.on('console', message => { if (message.type() === 'error') errors.push(message.text()); }); return errors;
}

test('岛链：空海真实候选取消无写，同候选保存刷新，追加岛链独立连通并保留旧陆地，逐次撤销', async ({ page }, info) => {

  const first = await previewIslands(page);
  expect(first).toMatchObject({ revision: 0, manifestDigest: blank.root.currentManifest, display: { source: 'generation-preview', revision: 1, verifiedBlocks: 256 } });

  await screenshot(page, info, 'islands-first-candidate-full-map.png', true);
  await panel(page, '创世'); await button(page, '取消生成'); await ready(page);
  expect(await saved(page, id)).toEqual(blank); expect(await databaseDigest(page)).toEqual(beforeDatabase); expect(page.url()).toBe(url);
  expect((await audit(page)).display).toMatchObject({ source: 'committed', manifestDigest: blank.root.currentManifest });
  const second = await previewIslands(page); expect(second.display.manifestDigest).toBe(first.display.manifestDigest);
  await button(page, '确认生成并保存'); await ready(page);
  const initial = await saved(page, id), initialData = await geography(page, initial.manifest), initialShape = inspect(initialData, initial.manifest.spec);
  expect(initial.root).toMatchObject({ revision: 1, currentManifest: first.display.manifestDigest, backupManifest: blank.root.currentManifest });
  expect(initial.manifest.generation.lastApplied).toMatchObject({ appliedRevision: 1, recipe: { generator: { id: 'island-chains', version: 1 }, seed: 'PH06-岛链-初生', parameters: { chainCount: 1, islandsPerChain: 3, radiusWorld: 80, reliefMeters: 1800 } } });
  expect(initialShape.componentCells).toHaveLength(3); for (const cells of initialShape.componentCells) expect(cells.length).toBeGreaterThanOrEqual(4);
  expect(initial.manifest.blockRefs.filter(ref => !ref.key.startsWith('elevation:'))).toEqual(blank.manifest.blockRefs.filter(ref => !ref.key.startsWith('elevation:')));
  for (let cell = 0; cell < initialData.heights.length; cell++) if (initialData.heights[cell] < 0) expect(initialData.heights[cell]).toBe(blankData.heights[cell]);
  expect(await refreshInNewDocument(page, id, initial, errors)).toEqual(initialData);
  const peakCell = initialData.heights.reduce((best, height, cell, heights) => height > heights[best] ? cell : best, 0); await near(page, peakCell); await screenshot(page, info, 'islands-saved-near-map.png');
  const buildResponse = await page.request.get('/build.json'); expect(buildResponse.ok()).toBe(true);
  const build = await buildResponse.json() as { islandGenerator: { sha256: string } };
  expect(initial.manifest.generation.lastApplied?.recipe.generator.fingerprint).toBe(build.islandGenerator.sha256); expect(build.islandGenerator.sha256).toMatch(/^[a-f0-9]{64}$/);
  const addition = await previewIslands(page, 'PH06-岛链-追加', '2', '64'); expect(addition.revision).toBe(1); expect(await saved(page, id)).toEqual(initial);
  await button(page, '确认生成并保存'); await ready(page);
  const added = await saved(page, id), addedData = await geography(page, added.manifest), addedShape = inspect(addedData, added.manifest.spec);
  expect(added.root).toMatchObject({ revision: 2, currentManifest: addition.display.manifestDigest, backupManifest: initial.root.currentManifest });
  expect(addedShape.componentCells).toHaveLength(5);
  const newComponents = addedShape.componentCells.filter(cells => cells.every(cell => initialData.heights[cell] < 0)); expect(newComponents).toHaveLength(2);
  for (const cells of newComponents) expect(cells.length).toBeGreaterThanOrEqual(4);
  let newLand = 0;
  for (let cell = 0; cell < addedData.heights.length; cell++) {
    if (initialData.heights[cell] >= 0 || addedData.heights[cell] < 0) expect(addedData.heights[cell]).toBe(initialData.heights[cell]);
    else newLand++;
  }
  expect(newLand).toBe(newComponents.reduce((sum, cells) => sum + cells.length, 0));
  expect(added.manifest.blockRefs.filter(ref => !ref.key.startsWith('elevation:'))).toEqual(initial.manifest.blockRefs.filter(ref => !ref.key.startsWith('elevation:')));
  expect(addedData.xyQ).toEqual(initialData.xyQ); expect(addedShape.worldAreaQ2).toBe(initialShape.worldAreaQ2);
  for (const cells of initialShape.componentCells) expect(addedShape.componentCells.some(component => component.length === cells.length && new Set(component).size === cells.length && cells.every(cell => component.includes(cell)))).toBe(true);
  await screenshot(page, info, 'islands-added-preserving-original-full-map.png', true);
  expect(await refreshInNewDocument(page, id, added, errors)).toEqual(addedData);
  await panel(page, '编辑'); await button(page, '撤销大世界编辑'); await ready(page);
  const undoneAddition = await saved(page, id); expect(undoneAddition.root.revision).toBe(3); expect(undoneAddition.manifest.blockRefs).toEqual(initial.manifest.blockRefs);
  await button(page, '撤销大世界编辑'); await ready(page);
