import { it } from 'vitest';
import { writeFileSync } from 'node:fs';
import { generateDetailShapeNatural } from '../../../src/world/generation/detail-shape-natural';

it('captures unchanged writer rectangle density failure once', async () => {
  const size = 128, centers = new Int32Array(size * size * 2), heights = new Int32Array(size * size).fill(-2400);
  for (let id = 0; id < heights.length; id++) { centers[id * 2] = (id % size * 32 + 16) * 256; centers[id * 2 + 1] = (Math.floor(id / size) * 32 + 16) * 256; }
  try {
    await generateDetailShapeNatural({ base: { format: 'lieguoji-empty-detail-shape-base', version: 1, size, centersQ: centers.buffer, elevationsMeters: heights.buffer, detailCoast: new Map(), locksPresent: false, entitiesPresent: false, generationPresent: false },
      recipe: { format: 'lieguoji-natural-shape-recipe', version: 2, seed: 'explicit-fine-new-land', coastStructureWorld: 16, coastRoughnessWorld: 0,
        reliefMeters: 2200, shelfWidthWorld: 96, islandChainCount: 0, islandsPerChain: 2,
        pointsQ: [[1024, 1024], [1664, 1024], [1664, 1536], [1024, 1536]].map(([x, y]) => ({ xQ: x * 256, yQ: y * 256 })) }
    }, { batchWork: 2048, yieldControl: async () => {} });
    throw new Error('Expected original density failure to reproduce.');
  } catch (error) {
    if (!(error instanceof Error) || !error.message.startsWith('DENSITY_DIAGNOSTIC:')) throw error;
    writeFileSync('.autopilot/ph06/fine-world-density-repair/density-before.json', JSON.stringify(JSON.parse(error.message.slice('DENSITY_DIAGNOSTIC:'.length)), null, 2));
  }
});
